"""
Inference — conforms to the GalaxEye I/O contract (Section 2.3).

CLI:
    python infer.py --input_dir <path> --output_dir <path> --weights <path/to/checkpoint>

Input  : directory of single-channel Sentinel-1 SAR (VV) patches, 256x256, 8-bit
         PNG, dB-scaled and min-max normalised to [0, 255].
Output : directory of generated 256x256 RGB PNG images, SAME filenames as inputs.

Design choices that satisfy the rest of the contract:
  - Runs on a single GPU with <=16 GB VRAM. The U-Net generator is small and we
    process one tile at a time in fp32, so memory is well under the budget. Falls
    back to CPU automatically if no GPU is present.
  - No internet at inference time. The generator architecture is rebuilt from the
    `model_cfg` stored *inside* the checkpoint, and there are no pretrained-backbone
    downloads anywhere in this file.
  - Channel adaptation: if the checkpoint was trained on multi-channel SAR
    (e.g. VV+VH from SEN12MS, input_nc=2), the single-channel VV input is repeated
    to match, as the contract allows.
"""

import argparse
import os

import numpy as np
import torch
from PIL import Image

from src.models import build_generator

IMG_EXTS = (".png", ".jpg", ".jpeg", ".tif", ".tiff")
TARGET = 256


def load_generator(weights_path, device):
    ckpt = torch.load(weights_path, map_location=device)
    if "model_cfg" not in ckpt or "generator" not in ckpt:
        raise ValueError(
            "Checkpoint missing 'model_cfg'/'generator'. Use a checkpoint saved "
            "by train.py (it embeds the architecture config)."
        )
    model_cfg = ckpt["model_cfg"]
    G = build_generator(model_cfg).to(device)
    G.load_state_dict(ckpt["generator"])
    G.eval()
    return G, model_cfg


def preprocess(path, input_nc):
    """8-bit single-channel PNG in [0,255] -> tensor in [-1,1], shape (1,C,256,256)."""
    img = Image.open(path).convert("L")
    if img.size != (TARGET, TARGET):
        img = img.resize((TARGET, TARGET), Image.BICUBIC)
    arr = np.asarray(img, dtype=np.float32)
    t = torch.from_numpy(arr).unsqueeze(0).unsqueeze(0) / 127.5 - 1.0   # (1,1,H,W)
    if input_nc > 1:                       # adapt single VV to multi-channel models
        t = t.repeat(1, input_nc, 1, 1)
    return t


def postprocess(tensor):
    """(1,3,H,W) in [-1,1] -> uint8 RGB array in [0,255]."""
    out = tensor.squeeze(0).detach().clamp(-1, 1).cpu().numpy()
    out = (out + 1.0) / 2.0
    out = np.transpose(out, (1, 2, 0))
    return (out * 255.0 + 0.5).astype(np.uint8)


@torch.no_grad()
def main():
    parser = argparse.ArgumentParser(description="SAR -> EO inference")
    parser.add_argument("--input_dir", required=True)
    parser.add_argument("--output_dir", required=True)
    parser.add_argument("--weights", required=True)
    args = parser.parse_args()

    device = "cuda" if torch.cuda.is_available() else "cpu"
    print(f"[infer] device = {device}")

    G, model_cfg = load_generator(args.weights, device)
    input_nc = model_cfg["input_nc"]

    os.makedirs(args.output_dir, exist_ok=True)
    files = sorted(f for f in os.listdir(args.input_dir)
                   if f.lower().endswith(IMG_EXTS))
    if not files:
        print(f"[infer] no images found in {args.input_dir}")
        return

    for fname in files:
        x = preprocess(os.path.join(args.input_dir, fname), input_nc).to(device)
        y = G(x)
        rgb = postprocess(y)
        # always write .png with the same stem, per the contract
        stem = os.path.splitext(fname)[0]
        Image.fromarray(rgb).save(os.path.join(args.output_dir, stem + ".png"))

    print(f"[infer] wrote {len(files)} images to {args.output_dir}")


if __name__ == "__main__":
    main()
